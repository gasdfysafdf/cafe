RPC 함수 (트랜잭션 단위로 묶인 게임 로직) -- --------------------------------------------------------------------- -- 5-1. 제조: 재고 차감 + 레시피 일치 여부 판정 (일치하면 menu_id, 아니면 0) CREATE OR REPLACE FUNCTION brew_drink( p_username VARCHAR, p_used INT[] ) RETURNS INT AS $$ DECLARE v_player UUID; v_menu_id INT; r RECORD; BEGIN SELECT id INTO v_player FROM players WHERE username = p_username; -- 재료별 사용량 집계 후 재고 차감 (재고 부족이면 전체 롤백) FOR r IN SELECT x AS ingredient_id, COUNT(*)::INT AS cnt FROM unnest(p_used) AS x GROUP BY x LOOP UPDATE ingredients SET stock = stock - r.cnt WHERE id = r.ingredient_id AND stock >= r.cnt; IF NOT FOUND THEN RAISE EXCEPTION '재고가 부족합니다 (재료 id %)', r.ingredient_id; END IF; END LOOP; -- 순서까지 일치하는 레시피 찾기 SELECT menu_id INTO v_menu_id FROM recipes GROUP BY menu_id HAVING array_agg(ingredient_id ORDER BY step_order) = p_used LIMIT 1; IF v_menu_id IS NULL THEN INSERT INTO sales_log (player_id, menu_id, success, earned, result) VALUES (v_player, NULL, FALSE, 0, 'wasted'); RETURN 0; END IF; RETURN v_menu_id; END; $$ LANGUAGE plpgsql; -- 5-2. 서빙: 돈 증가 + 판매 기록 (수입 반환) CREATE OR REPLACE FUNCTION serve_order( p_username VARCHAR, p_menu_id INT, p_tip INT DEFAULT 0 ) RETURNS INT AS $$ DECLARE v_player UUID; v_price INT; v_earned INT; BEGIN SELECT id INTO v_player FROM players WHERE username = p_username; SELECT price INTO v_price FROM menus WHERE id = p_menu_id; v_earned := v_price + GREATEST(p_tip, 0); UPDATE players SET money = money + v_earned WHERE id = v_player; INSERT INTO sales_log (player_id, menu_id, success, earned, result) VALUES (v_player, p_menu_id, TRUE, v_earned, 'served'); RETURN v_earned; END; $$ LANGUAGE plpgsql; -- 5-3. 손님이 화나서 떠남 기록 CREATE OR REPLACE FUNCTION customer_left( p_username VARCHAR, p_menu_id INT ) RETURNS VOID AS $$ BEGIN INSERT INTO sales_log (player_id, menu_id, success, earned, result) SELECT id, p_menu_id, FALSE, 0, 'angry_left' FROM players WHERE username = p_username; END; $$ LANGUAGE plpgsql; -- 5-4. 재료 구매: 잔액 검사 → 돈 차감 → 재고 증가 → 구매 기록 CREATE OR REPLACE FUNCTION buy_ingredient( p_username VARCHAR, p_ingredient_id INT, p_qty INT ) RETURNS INT AS $$ DECLARE v_player UUID; v_money INT; v_price INT; v_cost INT; BEGIN IF p_qty <= 0 THEN RAISE EXCEPTION '수량은 1개 이상이어야 합니다'; END IF; SELECT price INTO v_price FROM ingredients WHERE id = p_ingredient_id; v_cost := v_price * p_qty; SELECT id, money INTO v_player, v_money FROM players WHERE username = p_username FOR UPDATE; IF v_money < v_cost THEN RAISE EXCEPTION '돈이 부족합니다 (필요 %원 / 보유 %원)', v_cost, v_money; END IF; UPDATE players SET money = money - v_cost WHERE id = v_player; UPDATE ingredients SET stock = stock + p_qty WHERE id = p_ingredient_id; INSERT INTO purchase_log (player_id, ingredient_id, quantity, total_cost) VALUES (v_player, p_ingredient_id, p_qty, v_cost); RETURN v_money - v_cost; END; $$ LANGUAGE plpgsql; -- 5-5. 게임 초기화 (발표/시연용) CREATE OR REPLACE FUNCTION reset_game(p_username VARCHAR) RETURNS TEXT AS $$ DECLARE v_player UUID; BEGIN SELECT id INTO v_player FROM players WHERE username = p_username; DELETE FROM sales_log WHERE player_id = v_player; DELETE FROM purchase_log WHERE player_id = v_player; UPDATE players SET money = 5000 WHERE id = v_player; UPDATE ingredients SET stock = 2; RETURN 'reset done'; END; $$ LANGUAGE plpgsql; -- --------------------------------------------------------------------- -- 6. 뷰 (통계 화면에서 사용하는 JOIN / GROUP BY 쿼리) -- --------------------------------------------------------------------- CREATE OR REPLACE VIEW v_player_summary AS SELECT p.username, p.money, COALESCE((SELECT SUM(earned) FROM sales_log s WHERE s.player_id = p.id), 0) AS total_income, COALESCE((SELECT SUM(total_cost) FROM purchase_log b WHERE b.player_id = p.id), 0) AS total_spent, (SELECT COUNT(*) FROM sales_log s WHERE s.player_id = p.id AND s.success) AS served_count, (SELECT COUNT(*) FROM sales_log s WHERE s.player_id = p.id AND NOT s.success) AS fail_count FROM players p; CREATE OR REPLACE VIEW v_menu_stats AS SELECT m.id, m.name, COUNT(s.id) FILTER (WHERE s.success) AS sold, COUNT(s.id) FILTER (WHERE s.result = 'angry_left') AS angry_left, COALESCE(SUM(s.earned), 0) AS revenue FROM menus m LEFT JOIN sales_log s ON s.menu_id = m.id GROUP BY m.id, m.name; CREATE OR REPLACE VIEW v_daily_sales AS SELECT (created_at AT TIME ZONE 'Asia/Seoul')::date AS day, COUNT(*) FILTER (WHERE success) AS sold, COUNT(*) FILTER (WHERE NOT success) AS failed, COALESCE(SUM(earned), 0) AS revenue FROM sales_log GROUP BY 1 ORDER BY 1 DESC;
